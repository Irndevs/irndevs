Código da aula

# organiza_fotos.py — padroniza nomes de arquivos numa pasta
from pathlib import Path

pasta = Path("fotos")
for i, arq in enumerate(sorted(pasta.glob("*.jpg")), start=1):
    novo = pasta / f"foto_{i:03d}.jpg"
    arq.rename(novo)
    print(f"{arq.name} -&gt; {novo.name}")
