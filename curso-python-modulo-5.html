E-mails, arquivos e organização automática

Deixe a caixa de entrada e as pastas do computador se organizarem sozinhas, sem você mexer em nada.

O que você vai aprender

  • Enviar e-mails com anexos usando smtplib e MIME
  • Automatizar o disparo de relatórios por e-mail
  • Monitorar pastas em tempo real e organizar arquivos com watchdog

Código da aula

# email_relatorio.py — envia relatório por e-mail com anexo
import smtplib
from email.message import EmailMessage
from pathlib import Path

def enviar_relatorio(destinatario, caminho_anexo):
    msg = EmailMessage()
    msg["Subject"] = "Relatório automático"
    msg["From"] = "robo@irndevs.com"
    msg["To"] = destinatario
    msg.set_content("Segue em anexo o relatório gerado automaticamente.")

    anexo = Path(caminho_anexo)
    msg.add_attachment(
        anexo.read_bytes(),
        maintype="application",
        subtype="octet-stream",
        filename=anexo.name,
    )

    with smtplib.SMTP_SSL("smtp.exemplo.com", 465) as servidor:
        servidor.login("robo@irndevs.com", "senha_do_app")
        servidor.send_message(msg)

    print(f"E-mail enviado para {destinatario} com {anexo.name}")

if __name__ == "__main__":
    enviar_relatorio("time@empresa.com", "relatorio_consolidado.xlsx")
