Web scraping com Requests e BeautifulSoup

Colete dados de sites e transforme páginas HTML em informação estruturada, sem depender de ninguém copiar e colar.

O que você vai aprender

  • Fazer requisições HTTP com a biblioteca Requests e tratar erros de conexão
  • Extrair dados de páginas com BeautifulSoup e seletores CSS
  • Salvar os resultados coletados em um arquivo CSV pronto para análise

Código da aula

# scraper_cotacoes.py — coleta cotações de moedas e salva em CSV
import csv
import requests
from bs4 import BeautifulSoup

URL = "https://exemplo-financas.com/cotacoes"

def coletar_cotacoes():
    resposta = requests.get(URL, timeout=10)
    resposta.raise_for_status()

    sopa = BeautifulSoup(resposta.text, "html.parser")
    linhas = sopa.select("table#cotacoes tbody tr")

    dados = []
    for linha in linhas:
        moeda = linha.select_one(".moeda").get_text(strip=True)
        valor = linha.select_one(".valor").get_text(strip=True)
        dados.append({"moeda": moeda, "valor": valor})

    return dados

def salvar_csv(dados, caminho="cotacoes.csv"):
    with open(caminho, "w", newline="", encoding="utf-8") as arquivo:
        escritor = csv.DictWriter(arquivo, fieldnames=["moeda", "valor"])
        escritor.writeheader()
        escritor.writerows(dados)
    print(f"{len(dados)} cotações salvas em {caminho}")

if __name__ == "__main__":
    cotacoes = coletar_cotacoes()
    salvar_csv(cotacoes)
